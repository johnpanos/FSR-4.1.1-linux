cmake_minimum_required(VERSION 3.20)
project(scaling LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Vulkan REQUIRED)
add_library(fsr411 STATIC src/fsr411.cpp)
target_include_directories(fsr411 PUBLIC src deps/Vulkan-Headers/include)
target_link_libraries(fsr411 PUBLIC Vulkan::Vulkan)
add_executable(fsr411-bench src/bench.cpp)
target_link_libraries(fsr411-bench PRIVATE fsr411)
target_compile_options(fsr411 PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
target_compile_options(fsr411-bench PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
